\section*{Sets and Indices}

\begin{itemize}
    \item Components: $I=\{1,2,3\}$.
    \item Mission scenarios: $K=\{A,B\}$.
    \item Feasible spare-count values from the reliability table: $N=\{0,1,2,3,4,5\}$.
\end{itemize}

\section*{Parameters}

\begin{itemize}
    \item $c_i$: unit spare cost for component $i$ (code: \texttt{unit\_price[i-1]}). From \texttt{general\_parameters.csv},
    \[
    (c_1,c_2,c_3)=(20,30,40).
    \]
    \item $w_i$: unit spare weight for component $i$ (code: \texttt{unit\_weight[i-1]}). From \texttt{general\_parameters.csv},
    \[
    (w_1,w_2,w_3)=(2,4,6).
    \]
    \item $B_k$: total budget limit in scenario $k$. From \texttt{general\_parameters.csv},
    \[
    B_A=150,\qquad B_B=140.
    \]
    \item $W_k$: total weight limit in scenario $k$. From \texttt{general\_parameters.csv},
    \[
    W_A=20,\qquad W_B=18.
    \]
    \item $R_i(n)$: reliability of component $i$ when $n$ spares are chosen, read from \texttt{table\_1.csv}. The code stores these as \texttt{reliability[n][i-1]}. Explicitly,
    \[
    \begin{array}{c|cccccc}
    n & 0 & 1 & 2 & 3 & 4 & 5\\ \hline
    R_1(n) & 0.5 & 0.6 & 0.7 & 0.8 & 0.9 & 1.0\\
    R_2(n) & 0.6 & 0.75 & 0.95 & 1.0 & 1.0 & 1.0\\
    R_3(n) & 0.7 & 0.9 & 1.0 & 1.0 & 1.0 & 1.0
    \end{array}
    \]
    \item Scenario probabilities: $p_A=p_B=0.5$.
\end{itemize}

\section*{Decision Variables}

\begin{itemize}
    \item $s_i \in N$: number of spares selected for component $i$ (code: \texttt{s1}, \texttt{s2}, \texttt{s3} for $i=1,2,3$).
    \item $\mathrm{SysRel}_k$: system reliability in scenario $k$ (code: \texttt{sysrel\_A}, \texttt{sysrel\_B}).
    \item $\mathrm{ExpRel}$: expected system reliability (code: \texttt{expected\_rel}).
\end{itemize}

\section*{Objective}

Choose a single spare package shared by both scenarios to maximize expected system reliability:
\[
\max \ \mathrm{ExpRel}
\]
with
\[
\mathrm{SysRel}_A = R_1(s_1)\,R_2(s_2)\,R_3(s_3),
\]
\[
\mathrm{SysRel}_B = R_1(s_1)\,R_2(s_2)\,R_3(s_3),
\]
and
\[
\mathrm{ExpRel} = 0.5\,\mathrm{SysRel}_A + 0.5\,\mathrm{SysRel}_B.
\]

Equivalently, matching the implemented logic exactly,
\[
\max_{s_1,s_2,s_3 \in N}
\quad
0.5\prod_{i=1}^3 R_i(s_i) + 0.5\prod_{i=1}^3 R_i(s_i).
\]

\section*{Constraints}

The common spare package must satisfy both scenarios' logistics limits:
\[
\sum_{i=1}^3 c_i s_i \le B_A,
\]
\[
\sum_{i=1}^3 w_i s_i \le W_A,
\]
\[
\sum_{i=1}^3 c_i s_i \le B_B,
\]
\[
\sum_{i=1}^3 w_i s_i \le W_B,
\]
\[
s_i \in N \qquad \forall i \in I.
\]

Substituting the given data,
\[
20 s_1 + 30 s_2 + 40 s_3 \le 150,
\]
\[
2 s_1 + 4 s_2 + 6 s_3 \le 20,
\]
\[
20 s_1 + 30 s_2 + 40 s_3 \le 140,
\]
\[
2 s_1 + 4 s_2 + 6 s_3 \le 18,
\]
\[
s_1,s_2,s_3 \in \{0,1,2,3,4,5\}.
\]

\section*{Notes on Implementation}

\begin{itemize}
    \item The code does not formulate or solve an LP/MIP/NLP with an external optimizer. It performs exact finite enumeration over all triples
    \[
    (s_1,s_2,s_3)\in \{0,\dots,\max N\}^3,
    \]
    where \texttt{max\_spares = max(reliability.keys())}; in this instance, \(\max N=5\).
    \item A candidate triple is discarded unless each spare count appears as a key in the loaded reliability dictionary. Since the table contains all values \(0,\dots,5\), this reproduces the domain \(N=\{0,1,2,3,4,5\}\).
    \item Cost and weight are computed only for spares, not for the base units:
    \[
    \texttt{total\_cost}=\sum_i c_i s_i,\qquad
    \texttt{total\_weight}=\sum_i w_i s_i.
    \]
    \item Reliability is scenario-independent in the implementation, so \(\mathrm{SysRel}_A=\mathrm{SysRel}_B\) for every feasible solution. Nevertheless, the code explicitly computes
    \[
    \mathrm{ExpRel}=0.5\,\mathrm{SysRel}_A+0.5\,\mathrm{SysRel}_B.
    \]
    \item The search keeps the first feasible combination attaining the best objective value encountered in lexicographic enumeration order of \texttt{product(range(max\_spares + 1), repeat=3)}.
\end{itemize}
